## Checklist — Código Gerado por Agente

### 1. Integridade
- [ ] Código faz exatamente o que foi pedido (nem mais, nem menos)?
- [ ] Nenhuma dependência adicional não solicitada?
- [ ] Nenhuma chamada a API externa não autorizada?

### 2. Segurança Bancária
- [ ] Nenhum PII (CPF, nome, endereço) em logs, responses ou comentários?
- [ ] Credenciais em variáveis de ambiente, nunca hardcoded?
- [ ] Validação de input em todos os endpoints externos?

### 3. Qualidade de Testes
- [ ] Happy path coberto?
- [ ] Pelo menos 1 caso de erro/edge case?
- [ ] Testes verificam comportamento correto?

### 4. Auditabilidade
- [ ] Operações críticas têm audit log: userId, timestamp, operação, resultado?
- [ ] Audit log não contém dados sensíveis?

### 5. Produção
- [ ] Timeouts em todas as chamadas externas?
- [ ] Comportamento em falha tratado?

Decisão: [ ] APROVAR  [ ] APROVAR COM RESSALVAS  [ ] BLOQUEAR

health-check.sh — Verificação de Microserviços
Shell
Quando usar Verificar saúde de múltiplos microserviços antes de deploy ou em troubleshooting. 
#!/bin/bash
# health-check.sh — NTT Banking Microservices Health Check
# Uso: ./health-check.sh [ambiente: staging|prod]

ENVIRONMENT=${1:-staging}
TIMEOUT=5
SERVICES=(
  "auth-api:http://auth-api.${ENVIRONMENT}:3000/health"
  "payment-api:http://payment-api.${ENVIRONMENT}:3001/health"
  "account-api:http://account-api.${ENVIRONMENT}:3002/health"
  "notification-api:http://notification-api.${ENVIRONMENT}:3003/health"
)

echo "🔍 Health Check — Ambiente: ${ENVIRONMENT}"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

FAILED=0

for service_def in "${SERVICES[@]}"; do
  name="${service_def%%:*}"
  url="${service_def#*:}"

  response=$(curl -s -o /dev/null -w "%{http_code}" --max-time ${TIMEOUT} "${url}")

  if [ "$response" = "200" ]; then
    echo "✅ ${name}: OK"
  else
    echo "❌ ${name}: FALHA (HTTP ${response})"
    FAILED=$((FAILED + 1))
  fi
done

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

if [ $FAILED -gt 0 ]; then
  echo "⚠️  ${FAILED} serviço(s) com problema. Verifique antes do deploy."
  exit 1
else
  echo "✅ Todos os serviços saudáveis. Deploy pode prosseguir."
  exit 0
fi

pii-scan.sh — Scanner de PII no Repositório
Shell
Quando usar Detectar dados pessoais e credenciais acidentalmente commitados. 
#!/bin/bash
# pii-scan.sh — Scanner LGPD para repositórios NTT
# Uso: ./pii-scan.sh [diretório_raiz]

ROOT=${1:-.}
VIOLATIONS=0

echo "🔐 Scanner LGPD — ${ROOT}"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

# Padrão CPF (formato: XXX.XXX.XXX-XX ou XXXXXXXXXXX)
echo "📋 Verificando padrões de CPF..."
CPF_MATCHES=$(grep -rn --include="*.{ts,js,java,py,json,yaml,yml,env}" \
  -E "[0-9]{3}\.?[0-9]{3}\.?[0-9]{3}-?[0-9]{2}" \
  --exclude-dir={node_modules,.git,dist,build} "${ROOT}" 2>/dev/null)

if [ -n "$CPF_MATCHES" ]; then
  echo "⚠️  Padrões de CPF encontrados:"
  echo "$CPF_MATCHES" | head -10
  VIOLATIONS=$((VIOLATIONS + 1))
fi

# Secrets hardcoded
echo "🔑 Verificando secrets hardcoded..."
for pattern in "api_key\s*=\s*['\"]" "password\s*=\s*['\"]" "secret\s*=\s*['\"][^{]"; do
  MATCHES=$(grep -rn --include="*.{ts,js,java,py}" -iE "${pattern}" \
    --exclude-dir={node_modules,.git,dist} "${ROOT}" 2>/dev/null | \
    grep -v "process.env\|os.environ\|System.getenv\|// example\|# example")

  if [ -n "$MATCHES" ]; then
    echo "⛔ Possível secret hardcoded:"
    echo "$MATCHES"
    VIOLATIONS=$((VIOLATIONS + 1))
  fi
done

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
if [ $VIOLATIONS -gt 0 ]; then
  echo "❌ ${VIOLATIONS} categoria(s) com problemas LGPD. Revise antes do commit."
  exit 1
else
  echo "✅ Nenhum padrão de PII ou credencial encontrado."
  exit 0
fi

coverage-report.sh — Relatório de Cobertura de Testes
Shell
Quando usar Gerar relatório consolidado de cobertura de testes de múltiplos microserviços. 
#!/bin/bash
# coverage-report.sh — Relatório de cobertura multi-serviço
# Uso: ./coverage-report.sh

THRESHOLD=80
SERVICES=("auth-api" "payment-api" "account-api" "notification-api")
FAILED_SERVICES=()

echo "🧪 Relatório de Cobertura de Testes"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
printf "%-25s %-15s %-15s %s\n" "Serviço" "Cobertura" "Threshold" "Status"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

for service in "${SERVICES[@]}"; do
  if [ -d "./${service}" ]; then
    cd "./${service}"

    # Executar testes e extrair cobertura (Jest)
    COVERAGE=$(npm test -- --coverage --silent 2>/dev/null | \
      grep "All files" | awk '{print $4}' | tr -d '%')

    cd ..

    if [ -n "$COVERAGE" ]; then
      STATUS="✅ OK"
      if (( $(echo "$COVERAGE < $THRESHOLD" | bc -l) )); then
        STATUS="❌ FALHOU"
        FAILED_SERVICES+=("$service")
      fi
      printf "%-25s %-15s %-15s %s\n" "$service" "${COVERAGE}%" "${THRESHOLD}%" "$STATUS"
    else
      printf "%-25s %-15s %-15s %s\n" "$service" "N/A" "${THRESHOLD}%" "⚠️  Sem dados"
    fi
  fi
done

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
if [ ${#FAILED_SERVICES[@]} -gt 0 ]; then
  echo "❌ Serviços abaixo do threshold: ${FAILED_SERVICES[*]}"
  exit 1
else
  echo "✅ Todos os serviços acima de ${THRESHOLD}% de cobertura."
fi
